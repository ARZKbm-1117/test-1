        function calculateGrade(mark) {
            if (mark === "" || mark === null || isNaN(mark)) return "-";
            let m = parseFloat(mark);
            if (m >= 80) return "A";
            if (m >= 75) return "A-";
            if (m >= 70) return "B+";
            if (m >= 65) return "B";
            if (m >= 60) return "B-";
            if (m >= 55) return "C+";
            if (m >= 50) return "C";
            if (m >= 45) return "C-";
            if (m >= 40) return "D+";
            if (m >= 35) return "D";
            return "F";
        }

        function formatDriveImageUrl(url) {
            if (!url) return 'https://via.placeholder.com/150';
            if (url.includes("drive.google.com")) {
                let idMatch = url.match(/id=([a-zA-Z0-9_-]+)/) || url.match(/\/d\/([a-zA-Z0-9_-]+)/);
                if (idMatch && idMatch[1]) {
                    return `https://lh3.googleusercontent.com/d/${idMatch[1]}`;
                }
            }
            return url;
        }

        function getDriveEmbedUrl(url) {
            if (!url) return '';
            let idMatch = url.match(/id=([a-zA-Z0-9_-]+)/) || url.match(/\/d\/([a-zA-Z0-9_-]+)/);
            if (idMatch && idMatch[1]) {
                return `https://drive.google.com/file/d/${idMatch[1]}/preview`;
            }
            return url;
        }

        async function searchStudent() {
            const matric = document.getElementById("inputMatric").value.trim().toUpperCase();
            const alertBox = document.getElementById("searchAlert");
            const dashboard = document.getElementById("dashboardSection");

            if (!matric) {
                alert("Sila masukkan No. Matrik terlebih dahulu!");
                return;
            }

            alertBox.classList.add("hidden");

            try {
                const response = await fetch(`${SCRIPT_URL}?action=getStudent&matric=${encodeURIComponent(matric)}`);
                const result = await response.json();

                if (result.status === "success") {
                    currentStudentData = result.data;
                    if(!currentStudentData.setAnswerLinks) currentStudentData.setAnswerLinks = {};

                    const localData = JSON.parse(localStorage.getItem(`student_${matric}`));
                    if (localData && localData.setAnswerLinks) {
                        currentStudentData.setAnswerLinks = { ...currentStudentData.setAnswerLinks, ...localData.setAnswerLinks };
                    }

                    localStorage.setItem('lastSearchedMatric', matric);
                    saveToLocalStorage();

                    renderDashboard(currentStudentData);
                    dashboard.classList.remove("hidden");
                } else {
                    dashboard.classList.add("hidden");
                    alertBox.classList.remove("hidden");
                }
            } catch (err) {
                alert("Ralat semasa berhubung dengan database.");
                console.error(err);
            }
        }

        function saveToLocalStorage() {
            if (currentStudentData && currentStudentData.matric) {
                localStorage.setItem(`student_${currentStudentData.matric}`, JSON.stringify(currentStudentData));
            }
        }

            <button class="btn" style="padding: 6px 10px; font-size: 11px;" onclick="uploadSetAnswerPdf(${setIndex})">📤 Upload Jawapan Set ${setIndex}</button>
        </div>
        <div id="pdfStatus_set${setIndex}" style="margin-top: 4px; font-size: 11px;">
            ${hasPdf ? `<span style="color: green; font-weight: bold;">✅ Terwujud</span> <a href="${existingSetPdf}" target="_blank" class="btn-pdf-pro" style="padding: 2px 6px; font-size: 10px; margin-left: 4px;">${pdfSvgIcon} <span>Lihat File</span></a>` : ''}
        </div>
        <div id="preview_set${setIndex}" class="preview-container" style="${hasPdf ? 'display:block;' : 'display:none;'}">
            <iframe src="${hasPdf ? getDriveEmbedUrl(existingSetPdf) : ''}" id="iframe_set${setIndex}"></iframe>
        </div>
    </div>
`;
                }

                const row = document.createElement("div");
                row.className = "set-row";
                row.innerHTML = `
                    <div>
                        <strong>${labelText} (PER 100)</strong>
                        <div class="timestamp" id="time_col_${colIdx}">${scoreObj.time ? `Dikemaskini: ${scoreObj.time}` : 'Belum dimasukkan'}</div>
                    </div>
                    <div class="text-center">
                        <input type="number" id="input_col_${colIdx}" class="mark-input" min="0" max="100" 
                            value="${scoreObj.mark}" ${disabledAttr} ${lockTooltip}
                            onchange="autoSaveSingleMark(${colIdx})" oninput="updateGradeDisplay(${colIdx})" placeholder="0-100">
                    </div>
                    <div class="text-center">
                        <span id="grade_col_${colIdx}" class="grade-badge">${calculateGrade(scoreObj.mark)}</span>
                    </div>
                    <div class="action-buttons-cell text-center">${buttonsHtml}</div>
                    <div>${uploadHtml}</div>
                `;
                formContainer.appendChild(row);
            });
        }

        function updateGradeDisplay(colIdx) {
            const val = document.getElementById(`input_col_${colIdx}`).value;
            document.getElementById(`grade_col_${colIdx}`).innerText = calculateGrade(val);
        }

        async function autoSaveSingleMark(colIdx) {
            if (!currentStudentData) return;

            const inputVal = document.getElementById(`input_col_${colIdx}`).value;
            const calculatedGrade = calculateGrade(inputVal);
            const now = new Date().toLocaleString("ms-MY");

            showNotification("⏳ Auto-saving markah...");

            if (!currentStudentData.scores[colIdx]) currentStudentData.scores[colIdx] = {};
            currentStudentData.scores[colIdx].mark = inputVal;
            currentStudentData.scores[colIdx].grade = calculatedGrade;
            currentStudentData.scores[colIdx].time = now;

            const timeElem = document.getElementById(`time_col_${colIdx}`);
            if (timeElem) timeElem.innerText = `Dikemaskini: ${now}`;

            saveToLocalStorage();

            const payload = {
                matric: currentStudentData.matric.toUpperCase(),
                scores: currentStudentData.scores,
                setAnswerLinks: currentStudentData.setAnswerLinks || {}
            };

            try {
                const response = await fetch(SCRIPT_URL, {
                    method: "POST",
                    headers: { "Content-Type": "text/plain;charset=utf-8" },
                    body: JSON.stringify(payload)
                });
                const resData = await response.json();
                if (resData.status === "success") showNotification("✅ Markah & Gred tersimpan!");
                else showNotification("❌ Gagal simpan ke Sheet!");
            } catch (err) {
                showNotification("❌ Ralat sambungan!");
                console.error(err);
            }
        }

        async function uploadStudentAnswerPdf() {
            const fileInput = document.getElementById("pdfFileInput");
            const pdfStatus = document.getElementById("pdfStatus");
            const uploadBtn = document.getElementById("uploadPdfBtn");

            if (!fileInput.files || fileInput.files.length === 0) {
                alert("Sila pilih fail PDF jawapan pelajar terlebih dahulu!");
                return;
            }

            const file = fileInput.files[0];
            if (file.type !== "application/pdf") {
                alert("Sila pastikan fail dalam format PDF sahaja!");
                return;
            }

            pdfStatus.innerHTML = "⏳ Sedang memuat naik fail PDF ke Google Drive...";
            if (uploadBtn) uploadBtn.disabled = true;

            const reader = new FileReader();
            reader.onload = async function(e) {
                const arrayBuffer = e.target.result;
                const bytes = Array.from(new Uint8Array(arrayBuffer));

                const payload = {
                    matric: currentStudentData.matric,
                    uploadPdf: true,
                    setIndex: "trial",
                    fileBytes: bytes
                };

                try {
                    const response = await fetch(SCRIPT_URL, {
                        method: "POST",
                        headers: { "Content-Type": "text/plain;charset=utf-8" },
                        body: JSON.stringify(payload)
                    });
                    const res = await response.json();
                    if (res.status === "success") {
                        pdfStatus.innerHTML = `✅ Berjaya dimuat naik! <a href="${res.pdfUrl}" target="_blank" class="btn-pdf-pro" style="padding: 4px 10px; font-size: 12px; margin-left: 5px;">${pdfSvgIcon}<span>Lihat Jawapan (PDF)</span></a>`;
                        currentStudentData.answerPdfLink = res.pdfUrl;
                        saveToLocalStorage();
                        showNotification("✅ Jawapan Trial tersimpan!");
                    } else {
                        pdfStatus.innerHTML = "❌ Gagal memuat naik: " + (res.message || "Ralat");
                    }
                } catch (err) {
                    pdfStatus.innerHTML = "❌ Ralat memuat naik fail.";
                    console.error(err);
                } finally {
                    if (uploadBtn) uploadBtn.disabled = false;
                }
            };
            reader.readAsArrayBuffer(file);
        }

        async function uploadSetAnswerPdf(setIndex) {
            const fileInput = document.getElementById(`pdfFileInput_set${setIndex}`);
            const pdfStatus = document.getElementById(`pdfStatus_set${setIndex}`);

            if (!fileInput.files || fileInput.files.length === 0) {
                alert(`Sila pilih fail PDF jawapan Set ${setIndex} terlebih dahulu!`);
                return;
            }

            const file = fileInput.files[0];
            if (file.type !== "application/pdf") {
                alert("Sila pastikan fail dalam format PDF sahaja!");
                return;
            }

            pdfStatus.innerHTML = "⏳ Sedang memuat naik...";

            const reader = new FileReader();
            reader.onload = async function(e) {
                const arrayBuffer = e.target.result;
                const bytes = Array.from(new Uint8Array(arrayBuffer));

                const payload = {
                    matric: currentStudentData.matric,
                    uploadPdf: true,
                    setIndex: setIndex,
                    fileName: `Jawapan_Set${setIndex}_${currentStudentData.matric}.pdf`,
                    fileBytes: bytes
                };

                try {
                    const response = await fetch(SCRIPT_URL, {
                        method: "POST",
                        headers: { "Content-Type": "text/plain;charset=utf-8" },
                        body: JSON.stringify(payload)
                    });
                    const res = await response.json();
                    if (res.status === "success") {
                        const embedUrl = getDriveEmbedUrl(res.pdfUrl);
                        if(!currentStudentData.setAnswerLinks) currentStudentData.setAnswerLinks = {};
                        currentStudentData.setAnswerLinks[setIndex] = res.pdfUrl;
                        saveToLocalStorage();

                        pdfStatus.innerHTML = `<span style="color: green; font-weight: bold;">✅ Berjaya!</span> <a href="${res.pdfUrl}" target="_blank" class="btn-pdf-pro" style="padding: 2px 6px; font-size: 10px; margin-left: 4px;">${pdfSvgIcon} <span>Lihat File</span></a>`;
                        const previewBox = document.getElementById(`preview_set${setIndex}`);
                        const iframe = document.getElementById(`iframe_set${setIndex}`);
                        iframe.src = embedUrl;
                        previewBox.style.display = 'block';

                        showNotification(`✅ Fail Set ${setIndex} tersimpan!`);
                    } else {
                        pdfStatus.innerHTML = "❌ Gagal: " + (res.message || "Ralat");
                    }
                } catch (err) {
                    pdfStatus.innerHTML = "❌ Ralat sambungan.";
                    console.error(err);
                }
            };
            reader.readAsArrayBuffer(file);
        }
